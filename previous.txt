from flask import Flask, jsonify, send_from_directory
from gpiozero import Button

app = Flask(__name__)
pressure_mat = Button(17, pull_up=True, bounce_time=0.05)


@app.route('/')
def index():
    return send_from_directory('.', 'index.html')


@app.route('/pressure')
def pressure():
    return jsonify({
        'pressed': pressure_mat.is_pressed,
    })


app.run(host='0.0.0.0', port=5000)

<!DOCTYPE html>
<html>
<head>
    <title>Pressure Mat Test</title>

    <style>
        #status {
            width: 300px;
            height: 300px;
            background: red;
        }

        #status.active {
            background: green;
        }
    </style>
</head>

<body>
    <div id="status"></div>

    <script>
        const status = document.getElementById('status');

        async function checkPressureMat() {
            try {
                const response = await fetch('/pressure');
                const data = await response.json();

                status.classList.toggle('active', data.pressed);
            } catch (error) {
                console.error(error);
            }
        }

        setInterval(checkPressureMat, 100);
    </script>
</body>
</html>
